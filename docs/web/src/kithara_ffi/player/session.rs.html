2use crate::types::{FfiDuckingMode, FfiError, FfiInterruptionKind};

4/// Platform audio-session signals: route changes and competing sounds.
5#[cfg_attr(feature = "uniffi", uniffi::export)]
6impl AudioPlayer {
7    /// Notify the native player that the platform audio route changed.
8    ///
9    /// This does not change queue state. If playback is active, the
10    /// native output stream is recreated so CoreAudio/CPAL cannot keep a
11    /// stale route after headphones or Bluetooth devices are removed.
12    ///
13    /// # Errors
14    ///
15    /// Returns [`FfiError`] when the native player cannot schedule the
16    /// route invalidation.
17    pub fn notify_audio_route_changed(&self, reason: &str) -> Result<(), FfiError> {
18        self.inner.notify_audio_route_changed(reason)
19    }
20
21    /// Notify the native player that the platform interrupted, or released,
22    /// the audio output.
23    ///
24    /// An interruption stops the output below the engine: the audio callback
25    /// is no longer invoked, so playback can neither observe the interruption
26    /// nor report it, and every value the audio thread publishes freezes where
27    /// it stood. Reporting it here is what keeps the observable playback state
28    /// honest while nothing is audible. Getting the output back is
29    /// [`notify_audio_route_changed`](Self::notify_audio_route_changed).
30    pub fn notify_interruption(&self, kind: FfiInterruptionKind) {
31        self.inner.notify_interruption(kind.into());
32    }
33
34    /// Lower or restore the whole session output under a competing sound,
35    /// such as a call or a navigation prompt.
36    ///
37    /// # Errors
38    ///
39    /// Returns [`FfiError`] when the audio session rejects the change.
40    pub fn set_ducking_mode(&self, mode: FfiDuckingMode) -> Result<(), FfiError> {
41        self.inner.set_ducking_mode(mode)
42    }
43}
