15/// FFI-facing audio player. A thin facade over the platform-selected
16/// `Inner` engine (`NativeInner` on Apple / Android, `WasmInner` on
17/// wasm32). Every exported method delegates straight to `inner`; the
18/// facade only owns the object identity and (on native) the `Drop`
19/// shutdown pulse. The JS control surface lives in
20/// `crate::web::surface`.
21#[cfg_attr(feature = "uniffi", derive(uniffi::Object))]
22#[cfg_attr(target_arch = "wasm32", wasm_bindgen::prelude::wasm_bindgen)]
23pub struct AudioPlayer {
24    pub(crate) inner: Inner,
25}

324mod tests {
325    use std::sync::mpsc::{Receiver, Sender, channel};
326
327    use super::*;
328    use crate::types::FfiPlayerEvent;
329
330    struct ChannelObserver {
331        sender: Sender<FfiPlayerEvent>,
332    }
333
334    impl PlayerObserver for ChannelObserver {
335        fn on_event(&self, event: FfiPlayerEvent) {
336            self.sender.send(event).ok();
337        }
338    }
339
340    /// Blocks on the fact rather than on a deadline: the observer runs on the
341    /// event-bridge thread, so a local timeout would race it. A mode change
342    /// that never arrives is a hang, and the harness bounds hangs.
343    fn received_repeat_mode(receiver: &Receiver<FfiPlayerEvent>, expected: FfiRepeatMode) -> bool {
344        while let Ok(event) = receiver.recv() {
345            if matches!(
346                event,
347                FfiPlayerEvent::RepeatModeChanged { mode } if mode == expected
348            ) {
349                return true;
350            }
351        }
352        false
353    }
354
355    #[kithara::test]
356    fn repeat_mode_round_trips_and_notifies_observer() {
357        let player = AudioPlayer::new(FfiPlayerConfig::for_test()).expect("create player");
358        let (sender, receiver) = channel();
359        player.set_observer(Arc::new(ChannelObserver { sender }));
360
361        player
362            .set_repeat_mode(FfiRepeatMode::All)
363            .expect("valid repeat mode");
364
365        assert_eq!(player.repeat_mode(), FfiRepeatMode::All);
366        assert!(received_repeat_mode(&receiver, FfiRepeatMode::All));
367    }
