8/// Commands sent from the main-thread bridge to the engine Worker.
9///
10/// The multi-track queue variants
11/// (`Append`/`Insert`/`Remove`/`Replace`/`SelectQueue`/`RemoveAll`)
12/// mirror [`NativeInner`](crate::native::inner::NativeInner)'s queue
13/// methods — the caller allocates the [`TrackId`] on the main thread
14/// (shared-memory atomic, so ids stay process-monotonic across the
15/// worker boundary) and the worker plants it via `append_with_id` /
16/// `insert_with_id`.
17#[derive(Clone)]
18pub(crate) enum WorkerCmd {
19    Play,
20    Pause,
21    Stop,
22    Seek(f64),
23    SetVolume(f32),
24    SetCrossfade(CrossfadeSettings),
25    Next,
26    Previous,
27    SetEqGain {
28        band: u32,
29        gain_db: f32,
30    },
31    ResetEq,
32    /// Append a track to the tail of the queue. Loading starts in the
33    /// background; playback does not begin until a matching `SelectQueue`.
34    Append {
35        id: TrackId,
36        url: String,
37    },
38    /// Insert a track after `after` (or at the head when `after` is
39    /// `None`). Replies via `request_id` so the caller can observe the
40    /// `UnknownTrackId` rejection.
41    Insert {
42        id: TrackId,
43        url: String,
44        after: Option<TrackId>,
45        request_id: u32,
46    },
47    /// Remove a track by id. Replies via `request_id`.
48    Remove {
49        id: TrackId,
50        request_id: u32,
51    },
52    /// Replace the track at `index`. Replies via `request_id`.
53    Replace {
54        index: u32,
55        id: TrackId,
56        url: String,
57        request_id: u32,
58    },
59    /// Select (start playing) a queued track. Replies via `request_id`.
60    SelectQueue {
61        id: TrackId,
62        transition: Transition,
63        request_id: u32,
64    },
65    /// Clear every track from the queue.
66    RemoveAll,
67    /// Start (or restart) the analysis pass for a queued track, replying via
68    /// `request_id`.
69    Analyze {
70        id: TrackId,
71        request_id: u32,
72    },
73    /// Set the ABR mode on the worker's current track. `None` selects
74    /// automatic adaptation; `Some(index)` pins a manual variant. Mirrors
75    /// [`NativeInner::set_abr_mode`](crate::native::inner::NativeInner).
76    SetAbrMode {
77        variant_index: Option<u32>,
78    },
79    SetRepeat(RepeatMode),
80    SetPlaybackOrder(PlaybackOrder),
81    SetActionAtItemEnd(ActionAtItemEnd),
82    /// Lower or restore the whole session output through the owning Host.
83    SetDucking(SessionDuckingMode),
84    /// Apply per-network peak-bitrate ceilings to the worker's current ABR
85    /// handle. Mirrors
86    /// [`NativeInner::update_peak_bitrate`](crate::native::inner::NativeInner).
87    PeakBitrate {
88        wifi_bps: f64,
89        cellular_bps: f64,
90    },
91    /// Set or clear the player-wide auth token. An empty token removes the
92    /// `X-Auth-Token` header. Applied to subsequently built tracks. Mirrors
93    /// [`NativeInner::setup_network`](crate::native::inner::NativeInner).
94    AuthToken {
95        token: String,
96    },
97    /// Register a DRM key rule on the worker. The worker builds a
98    /// cross-thread key processor (the real JS callback stays on the main
99    /// thread) keyed on `salt`, then folds the rule into the player-wide
100    /// `KeyOptions` and header map.
101    SetupHlsAes {
102        salt: String,
103        domains: Vec<String>,
104        headers: Option<HashMap<String, String>>,
105        query_params: Option<HashMap<String, String>>,
106    },
107}
