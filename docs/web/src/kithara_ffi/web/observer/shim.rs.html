11/// Bridges a JS callback into the [`PlayerObserver`] trait. The wrapped
12/// [`Function`] is `!Send`; [`SendWrapper`] grants `Send + Sync` with a
13/// runtime same-thread assertion, so `Arc<dyn PlayerObserver>` keeps the
14/// trait's real `Send + Sync` supertraits on wasm without cfg.
15pub(crate) struct PlayerObserverJs {
16    func: SendWrapper<Function>,
17}
18
19impl PlayerObserverJs {
20    pub(crate) fn new(func: Function) -> Self {
21        Self {
22            func: SendWrapper::new(func),
23        }
24    }
25}
26
27impl PlayerObserver for PlayerObserverJs {
28    fn on_event(&self, event: FfiPlayerEvent) {
29        let _ = self
30            .func
31            .call1(&JsValue::UNDEFINED, &encode_player_event(&event));
32    }
33}

77/// Bridges a JS callback into the [`FfiKeyProcessor`] trait. The JS side
78/// is `process_key(key: Uint8Array, salt: string) -> Uint8Array`, invoked
79/// synchronously on the main thread by the
80/// [`key_processor_bridge`](crate::web::key_processor_bridge) pump. A
81/// non-`Uint8Array` return (or a throwing callback) yields an empty key,
82/// so the decrypt fails loudly downstream rather than silently using the
83/// raw bytes.
84pub(crate) struct KeyProcessorJs {
85    func: SendWrapper<Function>,
86}
87
88impl KeyProcessorJs {
89    pub(crate) fn new(func: Function) -> Self {
90        Self {
91            func: SendWrapper::new(func),
92        }
93    }
94}
95
96impl FfiKeyProcessor for KeyProcessorJs {
97    fn process_key(&self, key: Vec<u8>, salt: String) -> Vec<u8> {
98        let key_js = Uint8Array::from(key.as_slice());
99        let result = self.func.call2(
100            &JsValue::UNDEFINED,
101            key_js.as_ref(),
102            &JsValue::from_str(&salt),
103        );
104        result
105            .and_then(JsCast::dyn_into::<Uint8Array>)
106            .map_or_else(|_| Vec::new(), |arr| arr.to_vec())
107    }
108}
