20/// `BroadcastChannel` name carrying structured player events from the
21/// worker to the main-thread [`router`](crate::web::observer::router).
22pub(crate) const EVENT_CHANNEL: &str = "kithara-events";
23
24/// Subscribe to the queue's event bus inside the worker and forward every
25/// translated [`FfiPlayerEvent`] to the main thread over
26/// [`EVENT_CHANNEL`]. Spawned from
27/// [`worker_main`](crate::web::worker::worker_main).
28pub(crate) fn spawn(queue: &FfiQueueControl) {
29    let rx = queue.subscribe::<QueueBusEvent>();
30    let item_rx = queue.subscribe::<ItemBusEvent>();
31    task_spawn(async move {
32        run_items(item_rx).await;
33    });
34    task_spawn(async move {
35        run(rx).await;
36    });
37    spawn_duration_poll(queue);
38}

40/// Emit [`FfiPlayerEvent::DurationChanged`] whenever the current track's
41/// duration changes. `DurationChanged` is not a raw bus event: the native
42/// bridge derives it by polling [`FfiQueueControl::duration_seconds`], so the
43/// worker must do the same here. Without this the JS control surface never
44/// learns the track length and the seek slider has no range.
45fn spawn_duration_poll(queue: &FfiQueueControl) {
46    /// Poll cadence for the derived `DurationChanged` event, in milliseconds.
47    const DURATION_POLL_MS: u64 = 250;
48
49    let queue = queue.clone();
50    task_spawn(async move {
51        let Ok(channel) = BroadcastChannel::new(EVENT_CHANNEL) else {
52            return;
53        };
54        let mut last: Option<f64> = None;
55        loop {
56            if queue.is_closed() {
57                break;
58            }
59            let current = queue.duration_seconds();
60            if current != last
61                && let Some(seconds) = current
62            {
63                let _ = channel.post_message(&encode(&FfiPlayerEvent::DurationChanged { seconds }));
64                last = current;
65            }
66            sleep(Duration::from_millis(DURATION_POLL_MS)).await;
67        }
68    });
69}

71async fn run(mut rx: EventReceiver<QueueBusEvent>) {
72    let Ok(channel) = BroadcastChannel::new(EVENT_CHANNEL) else {
73        console::warn_1(&JsValue::from_str(
74            "kithara: BroadcastChannel unavailable in worker; event bridge disabled",
75        ));
76        return;
77    };
78    loop {
79        match rx.recv().await {
80            Ok(Envelope { event, .. }) => {
81                mirror_current_track(&event);
82                if let Some(ffi) = to_ffi(&event) {
83                    let _ = channel.post_message(&encode(&ffi));
84                }
85            }
86            Err(RecvError::Lagged(_)) => {}
87            Err(RecvError::Closed) => break,
88        }
89    }
90}

92/// Keep the main-thread current-track read-back
93/// ([`WorkerBridge::current_track_id`](crate::web::bridge::WorkerBridge))
94/// in sync by mirroring the worker's current-track cursor into the shared
95/// atomic on every relevant queue event.
96fn mirror_current_track(event: &QueueBusEvent) {
97    match event {
98        QueueBusEvent::Queue(QueueEvent::CurrentTrackChanged { id }) => {
99            crate::web::bridge::set_current_track_id(*id);
100        }
101        QueueBusEvent::Queue(QueueEvent::QueueEnded) => {
102            crate::web::bridge::set_current_track_id(None);
103        }
104        _ => {}
105    }
106}

108fn to_ffi(event: &QueueBusEvent) -> Option<FfiPlayerEvent> {
109    match event {
110        QueueBusEvent::Player(pe) => FfiPlayerEvent::try_from(pe).ok(),
111        QueueBusEvent::Queue(qe) => FfiPlayerEvent::try_from(qe).ok(),
112        _ => FfiPlayerEvent::try_from(event).ok(),
113    }
114}

116async fn run_items(mut rx: EventReceiver<ItemBusEvent>) {
117    let Ok(channel) = BroadcastChannel::new(EVENT_CHANNEL) else {
118        return;
119    };
120    loop {
121        match rx.recv().await {
122            Ok(Envelope { event, meta, .. }) => {
123                if let Ok(item_ffi) = crate::types::FfiItemEvent::try_from(&event)
124                    && let Some(track) = meta.track
125                {
126                    let msg = encode_item_event(&item_ffi);
127                    let _ = Reflect::set(
128                        &msg,
129                        &JsValue::from_str("scope"),
130                        &JsValue::from_str("item"),
131                    );
132                    let _ = Reflect::set(
133                        &msg,
134                        &JsValue::from_str("track_id"),
135                        &JsValue::from_f64(num_traits::cast(track.as_u64()).unwrap_or(0.0)),
136                    );
137                    let _ = channel.post_message(&msg);
138                }
139            }
140            Err(RecvError::Lagged(_)) => {}
141            Err(RecvError::Closed) => break,
142        }
143    }
