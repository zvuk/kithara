1//! Cross-platform FFI adapter for the kithara audio player.
2//!
3//! Wraps `kithara-play` types behind an FFI-friendly API. Native targets
4//! (Apple / Android) use `UniFFI` to generate Swift / Kotlin bindings; wasm32
5//! uses wasm-bindgen under `web`. `src/lib.rs` is the single
6//! structural boundary where target-conditional `cfg` gates live: shared
7//! FFI data types live in `core`, native-only bridges/runtime in `native`,
8//! and the wasm surface in `web`.
9
10#[cfg(all(feature = "uniffi", not(target_arch = "wasm32")))]
11uniffi::setup_scaffolding!();
12
13#[cfg(all(feature = "uniffi", not(target_arch = "wasm32")))]
14use kithara::events::TrackId;
15
16#[cfg(all(feature = "uniffi", not(target_arch = "wasm32")))]
17uniffi::custom_type!(TrackId, u64, { remote });
18
