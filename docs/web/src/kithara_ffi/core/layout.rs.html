36/// Foreign cache layout callback.
37///
38/// Implementations must be pure and deterministic, fast, non-blocking,
39/// non-throwing, and safe to call from arbitrary background threads. Returned
40/// values must not contain query text, credentials, or other secrets.
41/// `root` is called once for each store scope being created. `path` is called
42/// once for each resource key being minted. Cache operations using that key do
43/// not invoke either callback again.
44/// Invalid output fails scope or key creation and never falls back to the
45/// default layout.
46///
47/// `root` must return exactly one non-empty component and cannot equal
48/// `_index`. `path` must return a non-empty relative path of components
49/// separated by `/`; no component may end in `.tmp`. Components are ASCII,
50/// at most 96 bytes, never `.` or `..`, do not end in a dot or space, are not
51/// Windows device names, and contain neither control bytes nor
52/// `< > : " / \ | ? *`. Comparisons for `_index`, `.tmp`, and device names are
53/// case-insensitive. The store rejects invalid output instead of rewriting it.
54#[kithara::mock(api = FfiAssetLayoutMock)]
55#[cfg_attr(feature = "uniffi", uniffi::export(with_foreign))]
56pub trait FfiAssetLayout: Send + Sync {
57    fn path(&self, resource: FfiAssetResource) -> String;
58    fn root(&self, source: FfiAssetSource) -> String;
59}
